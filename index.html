        async function init() {

            await fetchOrdersFromCloud();

            // بروزرسانی خودکار هر ۴ ثانیه برای همگام‌سازی بین دستگاه‌ها

            setInterval(fetchOrdersFromCloud, 4000);

        async function fetchOrdersFromCloud() {

            try {

                const { data, error } = await supabaseClient.from('orders').select('*');

                if (!error && data) {

                    orders = data.map(item => item.data);

                    if (currentUser) renderDashboard();

                }

            } catch (err) {

                console.error("خطا در همگام‌سازی ابری:", err);

            }

        }

 

        async function saveOrderToCloud(order) {

            try {

                orders = orders.filter(o => o.id !== order.id);

                orders.push(order);

               

                await supabaseClient.from('orders').upsert({

                    id: order.id.toString(),

                    data: order,

                    updated_at: new Date()

                });

                renderDashboard();

            } catch (err) {

                console.error("خطا در ذخیره در سرور ابری:", err);

                alert("خطا در ثبت ابری. اتصال اینترنت خود را بررسی کنید.");

            }

        }

 

        function handleLogin() {

            const role = document.getElementById('role-select').value;

            const pin = document.getElementById('pin-input').value;

 

            if (pin === pins[role]) {

                currentUser = role;

                document.getElementById('login-section').classList.add('hidden');

                document.getElementById('dashboard-section').classList.remove('hidden');

                renderDashboard();

            } else {

                alert('رمز عبور اشتباه است! (پیش‌فرض مدیر: 1234، فروش: 1111، مالی: 2222، انبار: 3333، راننده: 4444)');

            }

        }

 

        function handleLogout() {

            currentUser = null;

            document.getElementById('dashboard-section').classList.add('hidden');

            document.getElementById('login-section').classList.remove('hidden');

            document.getElementById('pin-input').value = '';

        }

 

        function renderDashboard() {

                badge.className = 'px-3 py-1 bg-blue-600 text-white text-xs rounded-lg font-medium';

                container.innerHTML = renderSalesView();

            } else if (currentUser === 'finance') {

                badge.innerText = 'واحد مالی و حسابداری';

                badge.className = 'px-3 py-1 bg-amber-600 text-white text-xs rounded-lg font-medium';

                container.innerHTML = renderFinanceView();

                badge.innerText = 'انبارداری مرکزی';

                badge.className = 'px-3 py-1 bg-purple-600 text-white text-xs rounded-lg font-medium';

                container.innerHTML = renderWarehouseView();

                badge.innerText = 'کارتابل رانندگان';

        function createNewOrder() {

            const customer = document.getElementById('cust-name').value;

            const item = document.getElementById('item-desc').value;

            const qty = document.getElementById('item-qty').value;

            const price = document.getElementById('item-price').value;

 

            if (!customer || !item || !price) {

                alert('لطفاً تمام اطلاعات فاکتور را پر کنید.');

                return;

            }

 

            const newOrder = {

                id: Date.now(),

                customer,

                item,

                qty,

                price,

                status: 'pending_finance', // pending_finance -> pending_warehouse -> ready_to_ship -> delivered

                returnReason: '',

                date: new Date().toLocaleDateString('fa-IR')

            };

 

            saveOrderToCloud(newOrder);

            document.getElementById('cust-name').value = '';

            document.getElementById('item-desc').value = '';

            document.getElementById('item-price').value = '';

            alert('پیش‌فاکتور با موفقیت ثبت و به واحد مالی ارسال شد.');

        }

        function returnOrderToSales(id) {

            const reason = prompt('لطفاً دلیل عودت فاکتور به کارشناس فروش را وارد کنید (مثلاً کسری موجودی انبار / مغایرت پارت نامبر):');

            if (reason !== null) {

                const order = orders.find(o => o.id === id);

                if (order) {

                    order.status = 'returned';

                    order.returnReason = reason || 'بدون توضیح';

                    saveOrderToCloud(order);

                    alert('فاکتور با موفقیت به کارشناس فروش عودت داده شد.');

                }

            }

        }

        function deliverOrder(id) {

            const customerSign = prompt('نام تحویل‌گیرنده یا امضای دیجیتال مشتری را وارد کنید:');

            if (customerSign) {

                const order = orders.find(o => o.id === id);

                if (order) {

                    order.status = 'delivered';

                    order.signature = customerSign;

                    saveOrderToCloud(order);

                    alert('تحویل بار با موفقیت ثبت شد.');

                }

            }

        }

        async function updateOrderStatus(id, newStatus) {

            const order = orders.find(o => o.id === id);

            if (order) {

                order.status = newStatus;

                await saveOrderToCloud(order);

            }

        function getStatusBadge(status) {

            switch(status) {

                case 'pending_finance': return '<span class="px-2.5 py-1 bg-amber-950/60 text-amber-400 border border-amber-800/50 rounded-md text-[10px]">در انتظار مالی</span>';

                case 'pending_warehouse': return '<span class="px-2.5 py-1 bg-purple-950/60 text-purple-400 border border-purple-800/50 rounded-md text-[10px]">در صف انبار</span>';

                case 'ready_to_ship': return '<span class="px-2.5 py-1 bg-blue-950/60 text-blue-400 border border-blue-800/50 rounded-md text-[10px]">آماده بارگیری و ارسال</span>';

                case 'returned': return '<span class="px-2.5 py-1 bg-rose-950/60 text-rose-400 border border-rose-800/50 rounded-md text-[10px]">برگشت خورده به فروش</span>';

                default: return '<span class="px-2.5 py-1 bg-slate-700 text-slate-300 rounded-md text-[10px]">نامشخص</span>';

            }
